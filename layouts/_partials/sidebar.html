{{/* sidebar.html (site override) Forked from the compose theme's layouts/_partials/sidebar.html with two behavioral changes: 1. Root the tree at site.Home so every page sees the identical tree (same depths, sizes, indentation, and collapsed-state heuristics in custom.js). Upstream uses .FirstSection, which shifts the tree up a level as you navigate into a subsection. 2. Render a childless section bundle (e.g. tech/esphome/_index.md, created so the page can grow children later) as a leaf

link instead of an empty

group. Without this fix, page-bundle sections with no children render as section headings with nothing underneath, breaking the visual hierarchy. */}}
{{- template "tree" (dict "page" . "section" site.Home) }} {{- define "tree" }} {{- $section := .section }} {{- $page := .page }} {{- $permalink := $page.RelPermalink }} {{- $pages := $section.Pages }} {{- if $pages }} {{/* Section with children — render as a

group with a nested block of pages and (recursed) child sections. */}}

{{ $section.Title }}

{{- range $pages }} {{- if .IsPage }}

{{ .Title }}

{{- $toc_id := printf "toc-%s" (replace (urlize .Title) "." "-") -}} {{- $toc := string .TableOfContents }} {{- if ne $toc "" }} {{- safeHTML (replace $toc "TableOfContents" $toc_id) }} {{- end }} {{- else }} {{- template "tree" (dict "page" $page "section" .) }} {{- end }} {{- end }}
{{- else }} {{/* Section without children (page bundle used purely as a single leaf page — e.g. tech/esphome/_index.md). Render as a leaf

matching how individual .md pages render in their parent section, so the sidebar visual hierarchy stays consistent. */}}

{{ $section.Title }}

{{- end }} {{- end }}
