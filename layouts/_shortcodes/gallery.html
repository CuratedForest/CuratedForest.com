{{/* gallery.html — responsive thumbnail grid with a CSS-only lightbox. Usage (inside a page bundle that ships image resources): {{< gallery >}} # every image page resource {{< gallery match="*.jpg" sort="asc" thumb="400" full="1600" >}} Parameters (all optional): match : glob applied via .Resources.Match. Defaults to every image resource on the page. sort : "asc" (default) | "desc" — sort by resource Name. thumb : thumbnail width in CSS pixels (default 320). full : lightbox image width in pixels (default 1600). Capped at the original width by Hugo so portrait shots don't blow up beyond their native resolution. Lightbox mechanics: * No JavaScript. Each thumbnail is an and a sibling
sits at the end of the gallery. The :target pseudo-class flips the matching overlay from display:none to display:flex. * Backdrop and close button both point at "#_" (a no-op fragment that doesn't match any element) to close. Clicking outside the image hits the backdrop and closes; browser back also works. * Prev / next buttons inside the lightbox jump straight to the neighbour's id, threading through the gallery without ever unmounting the overlay. * The lightbox image is a Hugo-resized derivative (default 1600px wide) — lazy-loaded so 20+ images don't preload at gallery render time. */}} {{- $match := .Get "match" -}} {{- $sort := .Get "sort" | default "asc" -}} {{- $thumb := (.Get "thumb" | default "320") | int -}} {{- $full := (.Get "full" | default "1600") | int -}} {{- $resources := slice -}} {{- if $match -}} {{- $resources = .Page.Resources.Match $match -}} {{- else -}} {{- $resources = .Page.Resources.ByType "image" -}} {{- end -}} {{- if eq $sort "desc" -}} {{- $resources = sort $resources "Name" "desc" -}} {{- else -}} {{- $resources = sort $resources "Name" "asc" -}} {{- end -}} {{- $count := len $resources -}} {{- $galleryID := printf "g%s" (anchorize .Page.RelPermalink) -}} {{- if $resources -}}
{{- range $i, $r := $resources -}} {{- $thumbR := $r.Resize (printf "%dx" $thumb) -}} {{- $id := printf "%s-%d" $galleryID $i -}} {{- end -}}
{{- range $i, $r := $resources -}} {{- $fullR := $r.Resize (printf "%dx" $full) -}} {{- $id := printf "%s-%d" $galleryID $i -}} {{- $prevIdx := sub $i 1 -}} {{- if lt $prevIdx 0 }}{{ $prevIdx = sub $count 1 }}{{ end -}} {{- $nextIdx := add $i 1 -}} {{- if ge $nextIdx $count }}{{ $nextIdx = 0 }}{{ end -}} {{- $prevID := printf "%s-%d" $galleryID $prevIdx -}} {{- $nextID := printf "%s-%d" $galleryID $nextIdx -}}
{{- if gt $count 1 -}} ‹ › {{- end -}} ×
{{- end -}} {{- end -}}
